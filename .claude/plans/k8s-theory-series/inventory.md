## 재산출

원본이 늘거나 바뀌면 아래를 그대로 돌린다.

```bash
cd "/Users/jun/Documents/Obsidian Vault/☸️ k8s-theory/sessions"

# 편수
ls -1 *.md | wc -l

# 파일별 줄수 · 언어 미지정 코드블록 · wikilink
for f in *.md; do
  printf '%s\t%s\t%s\t%s\n' \
    "$(wc -l < "$f" | tr -d ' ')" \
    "$(( $(grep -c '^```$' "$f") / 2 ))" \
    "$(grep -o '\[\[[^]]*\]\]' "$f" | wc -l | tr -d ' ')" \
    "$f"
done | sort -n

# 언어가 지정된 코드블록 (원본엔 yaml 8개뿐이었다)
grep -ho '^```[a-z]*' *.md | grep -v '^```$' | sort | uniq -c

# 어투 — 격식체가 0이어야 원본 그대로다
grep -hoE '(습니다|입니다)' *.md | wc -l
```
